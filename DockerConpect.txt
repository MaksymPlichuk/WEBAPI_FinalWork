# Образи
docker images - переглянути список всіх завантажених Docker-образів.
docker pull <image_name> - завантажити образ з Docker Hub або іншого реєстру.
docker rmi <image_id_or_name> - видалити вказаний образ.
docker rmi -f <image_id_or_name> - примусово видалити образ, навіть якщо він використовується контейнером.
docker image prune - видалити всі "висячі" (dangling) образи, які не мають тегів.
docker tag <image_id> <new_image_name>:<new_tag> - додати новий тег до існуючого образу.
docker history <image> - переглянути історію шарів образу.
docker inspect <image> - переглянути детальну інформацію про образ.

# Контейнери
docker ps - переглянути список запущених контейнерів.
docker ps -a - переглянути список усіх контейнерів (як запущених, так і зупинених).
docker run <options> <image_name> - запустити новий контейнер з вказаним образом.
docker start <container_id_or_name> - запустити зупинений контейнер.
docker stop <container_id_or_name> - зупинити запущений контейнер.
docker rm <container_id_or_name> - видалити зупинений контейнер.
docker rm -f <container_id_or_name> - примусово видалити працюючий контейнер.
docker container prune - видалити всі зупинені контейнери.
docker logs <container_id_or_name> - перегляд журналу виведення контейнера.

# Options
-d 	 (для БД і image які діють на фоні)	- Запускає контейнер у фоновому режимі (detached mode)
-it 				- Інтерактивний режим з доступом до терміналу (-i + -t)
--name <name> 		- Присвоює ім’я контейнеру
--rm				- Видаляє контейнер після його завершення
-p HOST:CONTAINER	- Проброс порту (наприклад: -p 8080:80)

-v host:container	- Примонтований том (volume)
--network name		- Підключення до конкретної Docker-мережі
--restart always	- Автоматичний перезапуск контейнера (ще варіанти: no, on-failure, unless-stopped)
	
-e VARIABLE=value		- Встановлює змінну оточення
		Можна змінювати значення у файнлах (appsettings.json)

			-	"shop_nt"			
	
	api:
		Image: shop_api
		container_name: "shop_api"
		ports:
			-	"5000:8080"
